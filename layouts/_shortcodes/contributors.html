{{- /* 贡献者头像墙。数据来自 data/.yaml，不联网 —— 主题不在构建时打 GitHub API：那会让离线构建失败，也会把限流问题带进每一次 hugo server。 头像 URL 由数据文件自己给，要么是 GitHub 的 CDN，要么是本地图片。 墙的标记只有这一个调用方，直接写在这里；规范化那段有四类校验和去重， 单独成 partial。 markdown / rss 输出分支等 期 6 的输出格式落地再加 —— 现在没有 tdOutputFormat 这个 Store 键，凭空写一条读它的分支等于留一段跑不到的代码。 */ -}} {{- partial "validate-shortcode.html" (dict "shortcode" . "allowed" (slice "data" "class")) -}} {{- /* 数据文件名会进 hugo.Data 的索引，限成标识符：`../secrets` 那类值到不了 文件系统（index 只查内存里的 map），但一个带斜杠的键永远查不到东西， 早警告比渲染出一片空白好懂。 */ -}} {{- $key := "contributors" -}} {{- if and .IsNamedParams (isset .Params "data") -}} {{- $key = strings.TrimSpace (printf "%v" (.Get "data")) -}} {{- end -}} {{- if not (findRE `^[A-Za-z][A-Za-z0-9_-]*$` $key) -}} {{- warnf "shortcode %q: data must name one top-level data file at %s; got %q; rendering nothing" .Name .Position $key -}} {{- $key = "" -}} {{- end -}} {{- $items := slice -}} {{- with $key -}} {{- $data := index hugo.Data . -}} {{- if $data -}} {{- $items = partial "contributors/items.html" (dict "data" $data "key" .) -}} {{- else -}} {{- warnf "shortcode %q: data file %q was not found at %s; rendering nothing" $.Name . $.Position -}} {{- end -}} {{- end -}} {{- $class := "" -}} {{- if and .IsNamedParams (isset .Params "class") -}} {{- $class = strings.TrimSpace (printf "%v" (.Get "class")) -}} {{- if and $class (not (findRE `^[A-Za-z0-9_ -]+$` $class)) -}} {{- warnf "shortcode %q: class contains unsafe characters at %s; dropping the class" .Name .Position -}} {{- $class = "" -}} {{- end -}} {{- end -}} {{- with $items -}} {{- /* 标记走 contributors/wall.html：landing 的 contributors section 出同一 面墙，两套标记意味着 `.td-contributors` 的样式改一次要动两处。 */ -}} {{- partial "contributors/wall.html" (dict "items" . "id" (printf "contributors-%d" $.Ordinal) "class" $class) -}} {{- end -}}
