{{- /* 按键序列：`{{< kbd Ctrl Shift P >}}` → Ctrl + Shift + P 位置参数而非命名：按键是一个有序序列，`key1=` `key2=` 那种写法既啰嗦又 限制个数。 分隔的 `+` 是 aria-hidden 的视觉符号，另配一段仅供屏幕阅读器的文字 —— 朗读 "Ctrl with Shift with P" 才是这个序列的意思，读成 "Ctrl 加号 Shift" 不是。 */ -}} {{- partial "validate-shortcode.html" (dict "shortcode" . "form" "positional") -}} {{- $keys := slice -}} {{- /* 名字参数一律不认。校验那关已经警告过，这里照它说的忽略 —— 不加这层 判断，`{{< kbd key="Ctrl" >}}` 会照渲染，警告就成了空话。 */ -}} {{- if not .IsNamedParams -}} {{- range .Params -}} {{- $key := strings.TrimSpace (printf "%v" .) -}} {{- if $key -}} {{- $keys = $keys | append $key -}} {{- else -}} {{- warnf "shortcode %q: keys must not be empty at %s; skipping it" $.Name $.Position -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $keys -}} {{- warnf "shortcode %q needs at least one key at %s; rendering nothing" .Name .Position -}} {{- else if partial "content/plain.html" .Page -}} {{- /* 纯文本输出：`` `Ctrl`+`Shift`+`P` ``。用行内代码而不是纯文字，因为 Markdown 里没有 的对应写法，而按键名要与周围散文区分开。 `+` 直接写出来 —— 那段 sr-only 的 "with" 是给朗读器的，纯文本里 重复它只是噪声。 */ -}} {{- $parts := slice -}} {{- range $keys }}{{ $parts = $parts | append (printf "`%s`" .) }}{{ end -}} {{- delimit $parts "+" | safeHTML -}} {{- else -}} {{- range $index, $key := $keys -}} {{- if gt $index 0 -}} + {{ i18n "kbd_with" }} {{- end -}} {{ $key }} {{- end -}} {{- end -}}
