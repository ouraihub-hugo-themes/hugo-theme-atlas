{{- /* 脚本加载。 必须在正文渲染之后调用：render hook 与 shortcode 在渲染时往 Page Store 写能力标记，这里读它们决定加载哪些运行时。顺序反了标记还是空的，页面会 少加载它真正需要的东西。 每个运行时一个 bundle：一篇没有代码块的文档不该下载代码运行时。 */ -}} {{- $runtimes := slice "main" -}} {{- if .Store.Get "hasCodeRuntime" }}{{ $runtimes = $runtimes | append "code" }}{{ end -}} {{- if .Store.Get "hasTabs" }}{{ $runtimes = $runtimes | append "tabs" }}{{ end -}} {{- if .Store.Get "hasAssetList" }}{{ $runtimes = $runtimes | append "assets" }}{{ end -}} {{- /* 下面两个走 static/ 而不是 assets/dist/，因为它们开了代码分割：入口旁边 还有一堆 chunk，浏览器按入口里的相对路径直接取它们，不经过 Hugo 的资源 管线。放 assets/ 的话那些 chunk 一个都不会进 public/ —— 页面上是 404， 运行时加载到一半就断。 因此也没有 fingerprint 与 integrity：static/ 里的文件不是 resource。 chunk 名字自带内容哈希，换版本后旧 chunk 不会被缓存命中；入口本身一两 KB，它没有指纹的代价是一次 304。 */ -}} {{- if .Store.Get "hasMermaid" -}} {{- end -}} {{- if .Store.Get "hasCast" -}} {{- end -}} {{- range $runtimes -}} {{- with resources.Get (printf "dist/%s.js" .) -}} {{- if hugo.IsProduction -}} {{- $js := . | fingerprint -}} {{- else -}} {{- end -}} {{- end -}} {{- end -}}
