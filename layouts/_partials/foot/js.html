{{- /* 脚本加载。 必须在正文渲染之后调用：render hook 与 shortcode 在渲染时往 Page Store 写能力标记，这里读它们决定加载哪些运行时。顺序反了标记还是空的，页面会 少加载它真正需要的东西。 每个运行时一个 bundle：一篇没有代码块的文档不该下载代码运行时。 */ -}} {{- $runtimes := slice "main" -}} {{- if .Store.Get "hasCodeRuntime" }}{{ $runtimes = $runtimes | append "code" }}{{ end -}} {{- if .Store.Get "hasTabs" }}{{ $runtimes = $runtimes | append "tabs" }}{{ end -}} {{- if .Store.Get "hasAssetList" }}{{ $runtimes = $runtimes | append "assets" }}{{ end -}} {{- /* mermaid 走 static/ 而不是 assets/dist/，因为它是唯一开代码分割的 bundle： 入口旁边还有一百多个 chunk，浏览器按入口里的相对路径直接取它们，不经过 Hugo 的资源管线。放 assets/ 的话那些 chunk 一个都不会进 public/ —— 页面 上是 404，图一张都画不出来。 因此也没有 fingerprint 与 integrity：static/ 里的文件不是 resource。 chunk 名字自带内容哈希，换版本后旧 chunk 不会被缓存命中；入口本身只有 1.9 KB，它没有指纹的代价是一次 304。 */ -}} {{- if .Store.Get "hasMermaid" -}} {{- end -}} {{- range $runtimes -}} {{- with resources.Get (printf "dist/%s.js" .) -}} {{- if hugo.IsProduction -}} {{- $js := . | fingerprint -}} {{- else -}} {{- end -}} {{- end -}} {{- end -}}
