{{- /* 解析这一页的搜索参数，返回给 baseof 往 `
` 上挂属性。 两个键，都走 `.Param` 所以 cascade 继承成立（在 section 的 `_index.md` 里 写一次，整棵子树跟着走）： search_exclude 布尔。真则整页不进索引。 search_boost 正数。作者的意图是线性的"重要程度的倍数"。 **boost 要开平方再交给 pagefind。** 它的 `data-pagefind-weight` 是二次标度： 文档写明 2.0 约等于 4 倍影响、10.0 约等于 100 倍，取值范围 0.0–10.0。作者 写 `search_boost: 2` 想要的是"两倍重要"，直接透传会变成四倍。开平方之后 2 → 1.414，实际影响正好两倍。**不透传作者写的数字**，这是那条契约的要点。 于是 boost 的可用区间是 (0, 100]：100 开平方是 10，正好是 pagefind 的上限。 超出的钳到 100 并 warn —— 不静默钳，否则作者写 1000 与写 100 得到同样的 结果而页面上看不出来。 权重挂在正文包裹元素上（`
`），因为 pagefind 的元素权重是按元素算的， 而"这一页更重要"要作用于整页的正文。 返回 dict： exclude 布尔 attrs 直接往 main 上写的属性串（HTML 安全） */ -}} {{- $exclude := false -}} {{- with .Param "search_exclude" -}} {{- /* 只认真正的布尔。`search_exclude: "no"` 在 Go 模板里是非空字符串， 当真值处理会把整页从索引里去掉 —— 作者的意图恰好相反。 */ -}} {{- if eq (printf "%T" .) "bool" -}} {{- $exclude = . -}} {{- else -}} {{- warnidf (printf "search_exclude:%v" .) "%s: search_exclude must be a boolean; got %v; keeping the page indexed" $.Path . -}} {{- end -}} {{- end -}} {{- $weight := "" -}} {{- with .Param "search_boost" -}} {{- /* 判数字靠试算，不看类型名：`reflect` 没有 IsFloat/IsInt，而 `printf "%T"` 的结果取决于解析器（实测 YAML 里 42 是 uint64、-5 是 int64、3.8 是 float64）。照类型名单判，明天换个数就漏。 */ -}} {{- $probe := try (mul . 1.0) -}} {{- if $probe.Err -}} {{- warnidf (printf "search_boost:%v" .) "%s: search_boost must be a number; got %v; using 1.0" $.Path . -}} {{- else if le $probe.Value 0.0 -}} {{- /* 0 与负数不是"不重要"而是无意义：0 让整页的正文匹配不计分，那是 search_exclude 的活；负数在二次标度里没有定义。 */ -}} {{- warnidf (printf "search_boost:%v" .) "%s: search_boost must be greater than 0; got %v; using 1.0 (use search_exclude to drop a page)" $.Path . -}} {{- else -}} {{- $boost := $probe.Value -}} {{- if gt $boost 100.0 -}} {{- warnidf (printf "search_boost:%v" $boost) "%s: search_boost %v exceeds the maximum 100; clamping to 100" $.Path $boost -}} {{- $boost = 100.0 -}} {{- end -}} {{- /* 1.0 不输出属性：它就是默认值，写出来是噪声。 */ -}} {{- if ne $boost 1.0 -}} {{- $weight = printf "%.3f" (math.Sqrt $boost) -}} {{- end -}} {{- end -}} {{- end -}} {{- $attrs := "" -}} {{- if $exclude -}} {{- $attrs = " data-pagefind-ignore" -}} {{- else -}} {{- $attrs = " data-pagefind-body" -}} {{- with $weight -}}{{ $attrs = printf `%s data-pagefind-weight="%s"` $attrs . }}{{ end -}} {{- end -}} {{- return dict "exclude" $exclude "attrs" ($attrs | safeHTMLAttr) -}}
