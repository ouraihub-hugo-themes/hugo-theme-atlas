{{- /* 四个交叉引用目标（fig / tbl / eq / eg）共有的头部参数：num、id、caption、 class。四份一模一样的校验合成一处 —— 分开写的话，改一条 caption 规则要动 四个文件，而漏掉一个是静默的。 id 缺省是 `-`：`{{< tbl num="3" >}}` 的锚点就是 `#tbl-3`，作者 不必为每个目标起名字。num 校验不过时 id 变成 `tbl-`，登记器照样收 —— 一个 没有编号的目标仍然要能被列表看到，只是标签上没有数字。 class 直接丢弃不安全字符而不是清洗：一个被"清洗"过的类名会让作者以为它 生效了。字符集比块属性那条更宽一格（允许空格），因为这里收的是一整串。 optional 为真时 num 可以不给：那类组件（导图、图表围栏）不带编号也是完整的 内容，只是不进交叉引用表。这时返回的 num 与 id 都是空串，调用方据此决定要不 要登记。给了 id 却没给 num 是作者的误会 —— id 是编号目标的属性。 入参 dict：shortcode（传 `.`）、kind、optional 返回 dict：num、id、caption、class */ -}} {{- $sc := .shortcode -}} {{- $kind := .kind -}} {{- $optional := .optional | default false -}} {{- $at := $sc.Position -}} {{- $params := dict -}} {{- if $sc.IsNamedParams }}{{ $params = $sc.Params }}{{ end -}} {{- $hasNum := isset $params "num" -}} {{- if and (not $hasNum) (not $optional) -}} {{- warnf "shortcode %q requires parameter num at %s; rendering it unnumbered" $sc.Name $at -}} {{- else if and (not $hasNum) (isset $params "id") -}} {{- warnf "shortcode %q: id requires num at %s; ignoring it" $sc.Name $at -}} {{- end -}} {{- $num := "" -}} {{- $id := "" -}} {{- /* num 缺失且必需时也走一遍校验：那条路上 validate-num 返回空串，登记器照样 收 —— 一个没有编号的目标仍然要能被列表看到，只是标签上没有数字。 */ -}} {{- if or $hasNum (not $optional) -}} {{- $num = partial "book/validate-num.html" (dict "name" $sc.Name "value" ($sc.Get "num") "position" $at) -}} {{- $id = printf "%s-%s" $kind $num -}} {{- if isset $params "id" -}} {{- $id = partial "book/validate-id.html" (dict "name" $sc.Name "value" ($sc.Get "id") "position" $at) -}} {{- end -}} {{- end -}} {{- $caption := "" -}} {{- if isset $params "caption" -}} {{- $caption = $sc.Get "caption" -}} {{- if ne (printf "%T" $caption) "string" -}} {{- warnf "shortcode %q: caption must be a string at %s; dropping it" $sc.Name $at -}} {{- $caption = "" -}} {{- end -}} {{- $caption = strings.TrimSpace $caption -}} {{- end -}} {{- $class := "" -}} {{- if isset $params "class" -}} {{- $class = strings.TrimSpace (printf "%v" ($sc.Get "class")) -}} {{- if and $class (not (findRE `^[A-Za-z0-9_ -]+$` $class)) -}} {{- warnf "shortcode %q: class contains unsafe characters at %s; dropping the class" $sc.Name $at -}} {{- $class = "" -}} {{- end -}} {{- end -}} {{- return dict "num" $num "id" $id "caption" $caption "class" $class -}}
