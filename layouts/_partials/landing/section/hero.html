{{- /* Hero：landing 页最上面那一屏。 标题层级在这里是 h1，不是 head.html 的 h2 —— hero 的标题就是这一页的 标题，所以它自己出标题而不调那个共享件。一页有两个 hero 时第二个仍出 h1， 这是作者的问题，而校验它需要跨 section 的状态，代价高于收益。 `image` 是装饰性的：alt 恒为空。hero 图承载的信息在标题与副标题里已经说过 了，给它一段重复的 alt 只是让读屏软件把同一句话念两遍。真正承载信息的图 应该在正文里用 image render hook，那里 alt 是必需的。 键：headline、tagline、image、actions。用 headline 而不复用公共的 title： 公共 title 出 h2，而这里要 h1，两个键名分开比一个键在不同 section 里出 不同层级更少意外。 */ -}} {{- $page := .page -}} {{- $raw := .section -}} {{- $c := .common -}} {{- partial "landing/keys.html" (dict "section" $raw "at" $c.at "allowed" (slice "headline" "tagline" "image" "actions")) -}} {{- $headline := "" -}} {{- with index $raw "headline" -}} {{- $headline = $page.RenderString (strings.TrimSpace (printf "%v" .)) -}} {{- end -}} {{- $tagline := "" -}} {{- with index $raw "tagline" -}} {{- $tagline = $page.RenderString (dict "display" "block") (strings.TrimSpace (printf "%v" .)) -}} {{- end -}} {{- /* 尺寸要带上：没有 width/height 的 在载入前占 0 高，载入后把下面的 内容顶下去。hero 图大，这个抖动尤其刺眼。远程图拿不到固有尺寸（返回 0）， 那时不输出属性 —— 输出 width="0" 会把图压成不可见。 */ -}} {{- $image := dict "src" "" -}} {{- with index $raw "image" -}} {{- $image = partial "content/image-resolve.html" (dict "page" $page "src" (printf "%v" .) "at" $c.at) -}} {{- end -}}
