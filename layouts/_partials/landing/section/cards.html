"kind" "link" "parameter" "href" "position" $where "name" "landing card") -}} {{- $href = $policy.href -}} {{- $external = $policy.external -}} {{- end -}} {{- $icon := "" -}} {{- with index $item "icon" -}} {{- $icon = partial "content/icon.html" (dict "name" (strings.TrimSpace (printf "%v" .)) "class" "td-card-icon") -}} {{- end -}} {{- $image := false -}} {{- if isset $item "image" -}} {{- $image = partial "content/image-resolve.html" (dict "page" $page "src" (index $item "image") "position" $where "name" "landing card" "alt" (index $item "image_alt" | default "")) -}} {{- else if isset $item "image_alt" -}} {{- warnf "%s: image_alt needs image; ignoring it" $where -}} {{- end -}} {{- /* body 渲染 Markdown：它是描述，作者在里面写 `code` 与链接是常事。 */ -}} {{- $body := "" -}} {{- with index $item "body" -}} {{- with strings.TrimSpace (printf "%v" .) -}} {{- $body = $page.RenderString (dict "display" "block") . -}} {{- end -}} {{- end -}} {{- $cards = $cards | append (partial "content/card-markup.html" (dict "title" $title "body" $body "href" $href "external" $external "icon" $icon "badge" (strings.TrimSpace (printf "%v" (index $item "badge" | default ""))) "image" $image)) -}} {{- end -}} {{- end -}} {{- if $cards -}}
