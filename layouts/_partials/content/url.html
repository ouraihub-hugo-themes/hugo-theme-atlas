{{- /* 共享 URL 策略。图片、链接、以后的 shortcode 都过这一关。 这是安全边界，不是格式化工具。不合规的值一律丢弃并 warn，不做修补 —— 一个被"清洗"过的 URL 会让作者以为它生效了，而它指向的地方可能已经变了。 拒绝的四类： - 控制字符与空白：URL 里的换行能被用来伪造请求头 - 协议相对（//host）：跟随页面协议，HTTP 页面上会静默降级 - scheme 不在白名单：javascript: data: 是注入面 - http(s) 缺 host / mailto 缺地址：写错了，不是省略 入参 dict：value、kind（image | link）、parameter（消息里的参数名）、 position、name（消息里的组件名） 返回 dict：href、external、valid、raw、scheme、path */ -}} {{- $name := .name | default "content" -}} {{- $parameter := .parameter | default "URL" -}} {{- $position := .position | default "unknown position" -}} {{- $kind := .kind | default "link" -}} {{- $raw := strings.TrimSpace (printf "%v" (.value | default "")) -}} {{- $ok := true -}} {{- if not $raw -}} {{- warnf "%s: %s must not be empty at %s; dropping it" $name $parameter $position -}} {{- $ok = false -}} {{- else if findRE `[\x00-\x20\x7f]` $raw -}} {{- warnf "%s: %s must not contain whitespace or control characters at %s; dropping it" $name $parameter $position -}} {{- $ok = false -}} {{- else if hasPrefix $raw "//" -}} {{- warnf "%s: %s must not use a protocol-relative URL at %s; dropping it" $name $parameter $position -}} {{- $ok = false -}} {{- end -}} {{- $scheme := "" -}} {{- $urlPath := "" -}} {{- if $ok -}} {{- $parsed := try (urls.Parse $raw) -}} {{- if $parsed.Err -}} {{- warnf "%s: invalid %s %q at %s: %s; dropping it" $name $parameter $raw $position $parsed.Err -}} {{- $ok = false -}} {{- else -}} {{- $u := $parsed.Value -}} {{- $scheme = lower $u.Scheme -}} {{- $urlPath = $u.Path -}} {{- /* 图片没有 mailto 的道理。 */ -}} {{- $allowed := cond (eq $kind "image") (slice "http" "https") (slice "http" "https" "mailto") -}} {{- if and $scheme (not (in $allowed $scheme)) -}} {{- warnf "%s: unsupported %s scheme %q at %s (allowed: %s); dropping it" $name $parameter $scheme $position (delimit $allowed ", ") -}} {{- $ok = false -}} {{- else if and (in (slice "http" "https") $scheme) (not $u.Host) -}} {{- warnf "%s: %s %q requires a host at %s; dropping it" $name $parameter $raw $position -}} {{- $ok = false -}} {{- else if and (eq $scheme "mailto") (not (or $u.Opaque $u.Path)) -}} {{- warnf "%s: %s %q requires an address at %s; dropping it" $name $parameter $raw $position -}} {{- $ok = false -}} {{- else if and (eq $kind "image") (not $scheme) (not $u.Path) -}} {{- warnf "%s: %s %q requires an image path at %s; dropping it" $name $parameter $raw $position -}} {{- $ok = false -}} {{- end -}} {{- end -}} {{- end -}} {{- $href := "" -}} {{- if $ok -}} {{- $href = $raw -}} {{- /* 站内路径过 relURL，站点部署在子路径下才不会断。片段与查询串原样 保留 —— 它们相对当前页面，不该被加上前缀。 */ -}} {{- if and (not $scheme) (not (hasPrefix $raw "#")) (not (hasPrefix $raw "?")) -}} {{- $href = strings.TrimPrefix "/" $raw | relURL -}} {{- end -}} {{- end -}} {{- return dict "href" $href "external" (and $ok (ne $scheme "")) "valid" $ok "raw" (cond $ok $raw "") "scheme" (cond $ok $scheme "") "path" (cond $ok $urlPath "") -}}
