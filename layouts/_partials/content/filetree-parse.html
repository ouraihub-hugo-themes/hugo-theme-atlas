{{- /* 解析一段目录树文本成节点表。 认两种输入形状：`tree` 命令的输出（`├── │ └──`）与手写的缩进列表。作者 手里最常有的就是一段 `tree` 输出，要求他改写成别的形状等于让工具白跑。 深度靠缩进宽度的栈算，不靠除法：`tree` 的一级是 4 列，手写的可能是 2 格 或 tab，写死一个数会把另一种形状全算错。栈只记"见过哪些缩进宽度"，所以 两种形状都对，混用也对。 坏行跳过而不是整块丢弃：一段 tree 输出里粘进一行日志很常见。 入参 dict：source、position 返回 dict：nodes（每项含 depth/name/kind/children/…）、hasComments */ -}} {{- $position := .position | default "unknown position" -}} {{- $source := replace (printf "%s" .source) "\r" "" -}} {{- $lines := split (strings.TrimRight "\n" $source) "\n" -}} {{- $tones := slice "neutral" "info" "success" "warning" "danger" -}} {{- /* `\#` 要能表示字面的井号（文件名里有它，注释标记也是它）。先换成一个 正文里不可能出现的控制字符，切完注释再换回来。 下一行引号里是一个字面的 U+0001 字节，不是空串 —— 编辑器里看不见它。 改这一行时别把引号内容删掉当成清理空白。 */ -}} {{- $hashEscape := "" -}} {{- $nodes := slice -}} {{- $stack := slice -}} {{- $lineNo := 0 -}} {{- range $raw := $lines -}} {{- $lineNo = add $lineNo 1 -}} {{- $line := strings.TrimRight " \t" $raw -}} {{- if not (strings.TrimSpace $line) }}{{ continue }}{{ end -}} {{- /* `tree` 结尾那行统计（"3 directories, 7 files"）不是条目。 */ -}} {{- if findRE `^\s*\d+ director(?:y|ies)(?:, \d+ files?)?\s*$` $line }}{{ continue }}{{ end -}} {{- $width := 0 -}} {{- $rest := "" -}} {{- $tree := findRESubmatch "^((?:(?:│|\\|) | )*)(?:├──|└──|\\|--|`--) ?(.*)$" $line -}} {{- if $tree -}} {{- $prefix := index (index $tree 0) 1 -}} {{- $width = add (strings.RuneCount $prefix) 4 -}} {{- $rest = index (index $tree 0) 2 -}} {{- else -}} {{- $bullet := findRESubmatch `^(\s*)(?:[-*+]\s+)?(\S.*)$` $line -}} {{- if not $bullet -}} {{- warnf "filetree: line %d is not a tree entry at %s; skipping the line" $lineNo $position -}} {{- continue -}} {{- end -}} {{- /* tab 当 4 格：混用 tab 与空格时不换算会让同一层的两行算出不同宽度。 */ -}} {{- $indent := replace (index (index $bullet 0) 1) "\t" " " -}} {{- $width = strings.RuneCount $indent -}} {{- $rest = index (index $bullet 0) 2 -}} {{- end -}} {{- /* 缩进栈。比栈顶宽就进一层；窄就退到相等的那一层；退不到任何一层说明 这行的缩进从没开过（作者手抖多退了两格），跳过它 —— 硬塞进去会让 它下面的整棵子树都挂错父节点。 */ -}} {{- if not $stack -}} {{- $stack = slice $width -}} {{- else if gt $width (index $stack (sub (len $stack) 1)) -}} {{- $stack = $stack | append $width -}} {{- else -}} {{- range seq (len $stack) -}} {{- if and $stack (lt $width (index $stack (sub (len $stack) 1))) -}} {{- $stack = first (sub (len $stack) 1) $stack -}} {{- end -}} {{- end -}} {{- if or (not $stack) (ne $width (index $stack (sub (len $stack) 1))) -}} {{- warnf "filetree: line %d dedents to an indentation level that was never opened at %s; skipping the line" $lineNo $position -}} {{- continue -}} {{- end -}} {{- end -}} {{- $depth := sub (len $stack) 1 -}} {{- /* 先切注释，再取属性 —— 顺序反过来不行：属性段的正则锚在行尾，注释在 它后面就匹配不上，于是 `{tone=danger}` 整段落进名字里。 这也是对的顺序：作者写的是 `name {attrs} # 说明`，属性紧跟名字、说明 收尾。反过来还会让注释里的花括号（"见 {这里}"）被当成属性吃掉。 `#` 要么在行首要么前面有空白，否则 `a#b` 这种名字会被切开。 */ -}} {{- $comment := "" -}} {{- $escaped := replace $rest "\\#" $hashEscape -}} {{- with findRESubmatch `^(.*?)(?:^|\s+)#\s*(.*)$` $escaped -}} {{- $escaped = index (index . 0) 1 -}} {{- $comment = strings.TrimSpace (replace (index (index . 0) 2) $hashEscape "#") -}} {{- end -}} {{- $parsed := partial "content/line-attrs.html" (dict "line" (replace $escaped $hashEscape "#") "allowed" (slice "icon" "tone" "open" "type") "lineNo" $lineNo "kind" "filetree" "position" $position ) -}} {{- $attrs := $parsed.attrs -}} {{- $name := strings.TrimSpace $parsed.rest -}} {{- if not $name -}} {{- warnf "filetree: line %d has no entry name at %s; skipping the line" $lineNo $position -}} {{- continue -}} {{- end -}} {{- /* `[名字](链接)` 形式。链接过共享 URL 策略，不在这里另写一套判断。 */ -}} {{- $href := "" -}} {{- $external := false -}} {{- with findRESubmatch `^\[([^\]]+)\]\(([^)\s]+)\)$` $name -}} {{- $name = strings.TrimSpace (index (index . 0) 1) -}} {{- $url := partial "content/url.html" (dict "name" "filetree" "parameter" (printf "link on line %d" $lineNo) "position" $position "value" (index (index . 0) 2) "kind" "link" ) -}} {{- $href = $url.href -}} {{- $external = $url.external -}} {{- end -}} {{- /* 图标名过 icons.json —— 认不出的名字在这里 warn，比让 icon.html 在渲染时报一条不带行号的警告有用。 */ -}} {{- $icon := index $attrs "icon" | default "" -}} {{- if and $icon (not (isset (hugo.Data.icons | default dict) $icon)) -}} {{- warnf "filetree: line %d: unknown icon %q (see data/icons.json) at %s; using the default icon" $lineNo $icon $position -}} {{- $icon = "" -}} {{- end -}} {{- $tone := index $attrs "tone" | default "" -}} {{- if and $tone (not (in $tones $tone)) -}} {{- warnf "filetree: line %d: tone must be one of %s; got %q at %s; using neutral" $lineNo (delimit $tones " | ") $tone $position -}} {{- $tone = "" -}} {{- end -}} {{- $type := index $attrs "type" | default "" -}} {{- if and $type (not (in (slice "dir" "file") $type)) -}} {{- warnf "filetree: line %d: type must be dir or file; got %q at %s; inferring it" $lineNo $type $position -}} {{- $type = "" -}} {{- end -}} {{- $open := true -}} {{- $openSet := isset $attrs "open" -}} {{- if $openSet -}} {{- $value := index $attrs "open" -}} {{- if not (in (slice "true" "false") $value) -}} {{- warnf "filetree: line %d: open must be true or false; got %q at %s; leaving it open" $lineNo $value $position -}} {{- $openSet = false -}} {{- else -}} {{- $open = eq $value "true" -}} {{- end -}} {{- end -}} {{- $nodes = $nodes | append (dict "line" $lineNo "depth" $depth "name" $name "href" $href "external" $external "comment" $comment "icon" $icon "tone" $tone "open" $open "openSet" $openSet "type" $type ) -}} {{- end -}} {{- if not $nodes -}} {{- warnf "filetree: the fence has no entries at %s; rendering nothing" $position -}} {{- end -}} {{- /* 第二遍：有没有子节点、是目录还是文件。只有看过下一行才知道，所以 必须是第二遍 —— 第一遍里"下一行"还没解析。 */ -}} {{- $out := slice -}} {{- $hasComments := false -}} {{- $count := len $nodes -}} {{- /* 名字列宽在构建时按最长条目算出来，单位 ch。 这一项不是装饰：注释靠 grid 的第一列对齐，而每一层嵌套是一个独立的 `
    `，也就是一个独立的 grid 容器 —— `max-content` 只在同一个容器内 生效，跨层的行不会对齐。subgrid 要求所有行在同一个容器里，而树结构 本身就是嵌套的。所以列宽必须是一个所有行共用的具体数值。 缩进 2.5ch 与图标格 2ch + 间隙，与下面 CSS 里的 --td-filetree-indent 必须一致；改一处要改两处。 */ -}} {{- $nameWidth := 0.0 -}} {{- range $i, $node := $nodes -}} {{- $children := false -}} {{- if lt (add $i 1) $count -}} {{- $children = gt (index (index $nodes (add $i 1)) "depth") $node.depth -}} {{- end -}} {{- $kind := $node.type -}} {{- if not $kind -}} {{- /* 有子节点，或名字以 / 结尾 —— 后者让空目录也能标出来。 */ -}} {{- $kind = cond (or $children (hasSuffix $node.name "/")) "dir" "file" -}} {{- end -}} {{- if and $node.openSet (ne $kind "dir") -}} {{- warnf "filetree: line %d: open is only valid on directories at %s; ignoring it" $node.line $position -}} {{- end -}} {{- if $node.comment }}{{ $hasComments = true }}{{ end -}} {{- /* 缩进 + 名字 + 图标格。RuneCount 而非 len：CJK 文件名按字节算会宽出 三倍。等宽字体下 CJK 实际占两格，这里按一格算 —— 偏窄会让注释列被 挤，而 clamp 的 70% 上限兜住了它。 */ -}} {{- $width := add (add (mul 2.5 $node.depth) (strings.RuneCount $node.name)) 5 -}} {{- if gt $width $nameWidth }}{{ $nameWidth = $width }}{{ end -}} {{- $out = $out | append (merge $node (dict "kind" $kind "children" $children)) -}} {{- end -}} {{- $result := dict "nodes" $out "hasComments" $hasComments "nameWidth" $nameWidth -}} {{- return $result -}}
