{{- /* ```filetree {title="项目结构"} —— 一段目录树变成带图标的可折叠树。 用 `
    ` + `
    ` 而不是保留 `
    ` 里的画线字符：画线字符在屏幕
     阅读器里被逐个念出来（"竖线 横线 横线 src"），而列表嵌套本身就是"这个在
     那个里面"这个语义。折叠交给 `
    `，不写 JS。

     深度用 `--td-filetree-depth` 传给 CSS 而不是嵌套 padding：行要整行铺满
     （底色、hover 都靠它），缩进只作用于名字那一格。 */ -}}
{{- $page := .PageInner | default .Page -}}
{{- $policy := partial "content/attributes.html" (dict
  "attributes" .Attributes "allowed" (slice "title")
  "position" .Position "kind" "filetree"
) -}}
{{- $title := strings.TrimSpace (index $policy.consumed "title" | default "") -}}
{{- if and (isset $policy.consumed "title") (not $title) -}}
  {{- warnf "filetree: title must not be empty at %s; dropping it" .Position -}}
{{- end -}}

{{- $inner := printf "%s" .Inner -}}
{{- if not (strings.ContainsNonSpace $inner) -}}
  {{- warnf "filetree: the fence requires tree entries at %s; rendering nothing" .Position -}}
{{- else -}}
  {{- $parsed := partial "content/filetree-parse.html" (dict "source" $inner "position" .Position) -}}
  {{- $nodes := $parsed.nodes -}}
  {{- with $nodes -}}
    {{- $classes := slice "td-filetree" -}}
    {{- if not $parsed.hasComments }}{{ $classes = $classes | append "td-filetree--plain" }}{{ end -}}
    {{- with $policy.class }}{{ $classes = $classes | append . }}{{ end -}}
    {{- $count := len $nodes -}}
{{- /* 名字列宽只能是内联样式：它按这一段的内容算出来，每个围栏一个值。 */ -}}

    
  {{- with $title }}
    {{ . }}
    {{ end }}
  
      
    {{- range $i, $node := $nodes -}}
      {{- $row := partial "content/filetree-row.html" (dict
        "node" $node "hasComments" $parsed.hasComments
        "icons" (partial "content/filetree-icon.html" (dict
          "name" $node.name "kind" $node.kind "icon" $node.icon))
      ) -}}

      {{- /* 下一行的深度决定这一行之后要闭掉几层。范围里看不到"后面的行"，
             所以闭合标签就着 $nextDepth 在这里拼出来。最后一行的 nextDepth
             是 0，于是所有还开着的层都在它之后闭上。 */ -}}
      {{- $nextDepth := 0 -}}
      {{- if lt (add $i 1) $count }}{{ $nextDepth = (index $nodes (add $i 1)).depth }}{{ end -}}
      {{- $closes := sub $node.depth $nextDepth -}}
      {{- if $node.children }}{{ $closes = sub (add $node.depth 1) $nextDepth }}{{ end -}}

      {{- if $node.children -}}
        
    • {{ $row }}
        
      {{- else -}}
        
      • {{ $row }}
        • 
      {{- end -}}
      {{- if gt $closes 0 }}{{ range seq $closes }}
      • {{ end }}{{ end -}}
    {{- end }}
  
    

    
  {{- end -}}
{{- end -}}
