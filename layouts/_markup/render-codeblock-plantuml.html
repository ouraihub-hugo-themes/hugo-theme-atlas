{{- /* ```plantuml —— UML 图。构建时算出 URL，浏览器只发一个 ，零 JS。 PlantUML 的渲染器是 Java，没有能塞进浏览器的实现，所以图只能由服务器画。 这决定了两件事： 1. **默认关着。** 没配 params.ui.plantuml.server 就只渲染源码。指向 plantuml.com 作默认值会让每个装了主题的站点把作者的图发给第三方 —— 图里可能是内部系统的架构。 2. **零运行时。** 服务器接受十六进制编码的源码（`~h` 前缀，不压缩）， `printf "%x"` 就能算出来。既不需要 JS 也不需要 npm 依赖，图是一个 普通的 ，打印、无 JS、爬虫都一样在。 源码始终留在 
 里：服务器没配、图没加载出来、写错了三种情况下读者看到
     的是那段 UML 描述。它是给人读的纯文本。 */ -}}
{{- $page := .PageInner | default .Page -}}
{{- $policy := partial "content/attributes.html" (dict
  "attributes" .Attributes "allowed" (slice "caption" "num" "id" "alt")
  "position" .Position "kind" "plantuml"
) -}}
{{- $source := strings.TrimSpace (printf "%s" .Inner) -}}

{{- if not (strings.ContainsNonSpace $source) -}}
  {{- warnf "plantuml: the fence requires a diagram at %s; skipping the block" .Position -}}
{{- else -}}
  {{- $server := partial "content/plantuml-server.html" (dict "position" .Position) -}}
  {{- $caption := strings.TrimSpace (index $policy.consumed "caption" | default "") -}}
  {{- $alt := strings.TrimSpace (index $policy.consumed "alt" | default "") -}}
  {{- $numRaw := index $policy.consumed "num" | default "" -}}
  {{- $num := "" -}}
  {{- $id := "" -}}
  {{- if $numRaw -}}
    {{- $num = partial "book/validate-num.html" (dict "name" "plantuml" "value" $numRaw "position" .Position) -}}
    {{- $id = printf "fig-%s" $num -}}
    {{- with index $policy.consumed "id" -}}
      {{- $id = partial "book/validate-id.html" (dict "name" "plantuml" "value" . "position" $.Position) -}}
    {{- end -}}
    {{- partial "book/register-target.html" (dict
      "page" $page "kind" "fig" "num" $num "id" $id "caption" $caption
      "position" .Position "ordinal" .Ordinal
    ) -}}
  {{- else if isset $policy.consumed "id" -}}
    {{- warnf "plantuml: id requires num at %s; ignoring it" .Position -}}
  {{- end -}}

  {{- /* @startuml/@enduml 可以省：作者写一行 `Alice->Bob: hi` 也该出图，而
         服务器要求成对的定界符。补上比让作者记住它更省事。 */ -}}
  {{- $wrapped := $source -}}
  {{- if not (findRE `(?m)^\s*@start` $source) -}}
    {{- $wrapped = printf "@startuml\n%s\n@enduml" $source -}}
  {{- end -}}

  {{- /* 十六进制编码，`~h` 前缀。不压缩，所以 URL 随图变长 —— 一张几十行的
         图约两千字符，仍在所有浏览器与服务器的 URL 长度限制之内。选它而不选
         默认的 deflate+自定义 base64，是因为后者在 Hugo 模板里没有实现路径：
         没有 deflate 函数，也没有能改字母表的 base64。 */ -}}
  {{- $encoded := printf "~h%x" $wrapped -}}
  {{- $src := "" -}}
  {{- with $server.server -}}
    {{- $src = printf "%s/%s/%s" . $server.format $encoded -}}
  {{- end -}}


  {{- with $src }}
  {{- /* loading=lazy + decoding=async：图在服务器那边渲染，慢一点不该挡住正文。
         没有宽高，因此给一个最小高度占位，避免图到达时把正文推下去。 */}}
  
  {{- end }}
  
{{ $source }}

  {{- with $caption }}
  
  {{- end }}

{{- end -}}
