{{- /* 代码块。 高亮本身交给 Hugo 的 transform.Highlight —— 行号、高亮行、起始行号都是 它的 options，不重新实现。这里只加 Hugo 不做的四件事：文件名标题栏、 复制按钮、超长折叠、软换行。 入参属性：filename、copy、collapse、wrap、id、class、data-*、aria-* 高亮 options（lineNos、hl_lines、linenostart…）走围栏本身的语法， 不经过属性策略。 */ -}} {{- $page := .PageInner | default .Page -}} {{- $policy := partial "content/attributes.html" (dict "attributes" .Attributes "allowed" (slice "id" "filename" "copy" "collapse" "wrap") "position" .Position "kind" "code block") -}} {{- $code := strings.TrimSuffix "\n" .Inner -}} {{- $lang := lower (.Type | default "") -}} {{- $filename := strings.TrimSpace (index $policy.consumed "filename" | default "") -}} {{- $id := strings.TrimSpace (index $policy.consumed "id" | default "") -}} {{- /* shortcode body 是独立文档，.Ordinal 从 0 重数 —— 页面上的第一个围栏和 卡片里的第一个围栏都会算出 code-0。作用域前缀由 content/render-block.html 在渲染前写进 Store。 */ -}} {{- if not $id }}{{ $id = printf "code-%s%d" ($page.Store.Get "tdBlockScope" | default "") .Ordinal }}{{ end -}} {{- /* 布尔属性统一走 validate-bool：作者写 copy="yes" 时要有一条明确的警告， 而不是静默当成 false。 */ -}} {{- $copy := partial "validate-bool.html" (dict "value" (index $policy.consumed "copy") "fallback" true "key" "code block copy" "position" .Position) -}} {{- $wrap := partial "validate-bool.html" (dict "value" (index $policy.consumed "wrap") "fallback" false "key" "code block wrap" "position" .Position) -}} {{- /* collapse 接受布尔或正整数：true 用默认行数阈值，数字给自定义阈值。 这是"多设置特性也接受布尔简写"的那一类，不是普通开关。 */ -}} {{- $collapseLines := 0 -}} {{- with index $policy.consumed "collapse" -}} {{- $raw := strings.TrimSpace . -}} {{- if findRE `^[1-9][0-9]*$` $raw -}} {{- $collapseLines = int $raw -}} {{- else if in (slice "true" "false") (lower $raw) -}} {{- if eq (lower $raw) "true" }}{{ $collapseLines = 20 }}{{ end -}} {{- else -}} {{- warnf "code block: collapse must be true, false, or a positive integer; got %q at %s; not collapsing" $raw .Position -}} {{- end -}} {{- end -}} {{- $highlighted := transform.Highlight $code $lang .Options -}} {{- /* 外壳与 demo 的源码那半共用 —— 复制按钮、折叠、标题栏在两处必须是同一份， 否则示例页里的源码块会慢慢缺功能。软换行与 lntable 的互斥判断也在里面， 它看的是高亮结果。 */ -}} {{- partial "content/code-shell.html" (dict "page" $page "highlighted" $highlighted "code" $code "lang" $lang "id" $id "filename" $filename "copy" $copy "wrap" $wrap "collapseLines" $collapseLines "class" $policy.class "generic" (partial "content/attrs.html" $policy.generic)) -}}
